    32→func NewFileResource() resource.Resource {
    33→	return &FileResource{
    34→		sshClientFactory: DefaultSSHClientFactory,
    35→	}
    36→}
    37→
    38→// SSHClientFactory is a function type that creates SSH clients.
    39→// This allows for dependency injection in tests.
    40→type SSHClientFactory func(config gosftp.Config) (gosftp.ClientInterface, error)
    41→
    42→// DefaultSSHClientFactory creates real SSH clients.
    43→var DefaultSSHClientFactory SSHClientFactory = func(config gosftp.Config) (gosftp.ClientInterface, error) {
    44→	return gosftp.NewClient(config)
    45→}

   379→func (r *FileResource) Create(ctx context.Context, req resource.CreateRequest, resp *resource.CreateResponse) {
   380→	var data FileResourceModel
   381→
   382→	resp.Diagnostics.Append(req.Plan.Get(ctx, &data)...)
   383→	if resp.Diagnostics.HasError() {
   384→		return
   385→	}
   386→
   387→	ctx = tflog.SetField(ctx, "host", data.Host.ValueString())
   388→	ctx = tflog.SetField(ctx, "source", data.Source.ValueString())
   389→	ctx = tflog.SetField(ctx, "destination", data.Destination.ValueString())
   390→
   391→	tflog.Info(ctx, "Creating file resource")
   392→
   393→	// Calculate source file hash.
   394→	hash, size, err := hashFile(data.Source.ValueString())
   395→	if err != nil {
   396→		resp.Diagnostics.AddError("Failed to read source file", err.Error())
   397→		return
   398→	}
   399→	tflog.Debug(ctx, "Computed source file hash", map[string]interface{}{
   400→		"hash": hash,
   401→		"size": size,
   402→	})
   403→
   404→	// Create SSH client.
   405→	tflog.Debug(ctx, "Establishing SSH connection")
   406→	client, err := r.createSSHClient(&data)
   407→	if err != nil {
   408→		resp.Diagnostics.AddError("Failed to create SSH connection", err.Error())
   409→		return
   410→	}
   411→	defer r.releaseSSHClient(&data, client)
   412→	tflog.Debug(ctx, "SSH connection established")
   413→
   414→	// Upload file.
   415→	tflog.Debug(ctx, "Uploading file")
   416→	if err := client.UploadFile(ctx, data.Source.ValueString(), data.Destination.ValueString()); err != nil {
   417→		resp.Diagnostics.AddError("Failed to upload file", err.Error())
   418→		return
   419→	}
   420→	tflog.Debug(ctx, "File uploaded successfully")
   421→
   422→	// Set ownership and permissions.
   423→	tflog.Debug(ctx, "Setting file attributes", map[string]interface{}{
   424→		"owner": data.Owner.ValueString(),
   425→		"group": data.Group.ValueString(),
   426→		"mode":  data.Mode.ValueString(),
   427→	})
   428→	if err := client.SetFileAttributes(
   429→		ctx,
   430→		data.Destination.ValueString(),
   431→		data.Owner.ValueString(),
   432→		data.Group.ValueString(),
   433→		data.Mode.ValueString(),
   434→	); err != nil {
   435→		resp.Diagnostics.AddError("Failed to set file attributes", err.Error())
   436→		return
   437→	}
   438→
   439→	// Set computed values.
   440→	data.ID = types.StringValue(fmt.Sprintf("%s:%s", data.Host.ValueString(), data.Destination.ValueString()))
   441→	data.SourceHash = types.StringValue(hash)
   442→	data.Size = types.Int64Value(size)
   443→
   444→	tflog.Info(ctx, "File resource created successfully", map[string]interface{}{
   445→		"id":   data.ID.ValueString(),
   446→		"size": size,
   447→	})
   448→
   449→	resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
   450→}

   452→func (r *FileResource) Read(ctx context.Context, req resource.ReadRequest, resp *resource.ReadResponse) {
   453→	var data FileResourceModel
   454→
   455→	resp.Diagnostics.Append(req.State.Get(ctx, &data)...)
   456→	if resp.Diagnostics.HasError() {
   457→		return
   458→	}
   459→
   460→	ctx = tflog.SetField(ctx, "id", data.ID.ValueString())
   461→	tflog.Debug(ctx, "Reading file resource state")
   462→
   463→	// Read refreshes state from the resource.
   464→	// IMPORTANT: We do NOT update source_hash here. The source_hash represents
   465→	// what was last successfully synced to the remote, not the current local file.
   466→	// This allows Terraform to detect local file changes during plan.
   467→	//
   468→	// The plan modifier on source_hash computes the current local file hash
   469→	// and triggers an update when it differs from the stored state hash.
   470→
   471→	sourcePath := data.Source.ValueString()
   472→
   473→	// If source is not set (e.g., after import), don't remove from state.
   474→	// The user needs to update their config to include the source attribute.
   475→	if sourcePath == "" || data.Source.IsNull() || data.Source.IsUnknown() {
   476→		tflog.Debug(ctx, "Source path not set (possibly after import), preserving state")
   477→		// Preserve the current state - source will be set in next apply.
   478→		resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
   479→		return
   480→	}
   481→
   482→	// Check if this is a post-import situation where we need to sync local from remote.
   483→	// Conditions:
   484→	// 1. import_syncs_local is true
   485→	// 2. source_hash is empty/null (indicates fresh import or never synced)
   486→	// 3. Local file doesn't exist OR differs from what we expect
   487→	localFileExists := true
   488→	if _, err := os.Stat(sourcePath); os.IsNotExist(err) {
   489→		localFileExists = false
   490→	}
   491→
   492→	shouldSyncFromRemote := !data.ImportSyncsLocal.IsNull() &&
   493→		data.ImportSyncsLocal.ValueBool() &&
   494→		(data.SourceHash.IsNull() || data.SourceHash.ValueString() == "") &&
   495→		!localFileExists
   496→
   497→	if shouldSyncFromRemote {
   498→		tflog.Info(ctx, "Post-import sync: downloading remote file to local source path")
   499→		if err := r.syncLocalFromRemote(ctx, &data); err != nil {
   500→			resp.Diagnostics.AddError(
   501→				"Failed to sync local file from remote",
   502→				fmt.Sprintf("import_syncs_local is enabled but sync failed: %v", err),
   503→			)
   504→			return
   505→		}
   506→
   507→		// After syncing, compute the hash of the now-local file and update state.
   508→		hash, size, err := hashFile(sourcePath)
   509→		if err != nil {
   510→			resp.Diagnostics.AddError(
   511→				"Failed to compute hash after sync",
   512→				fmt.Sprintf("Could not hash synced file %s: %v", sourcePath, err),
   513→			)
   514→			return
   515→		}
   516→
   517→		data.SourceHash = types.StringValue(hash)
   518→		data.Size = types.Int64Value(size)
   519→
   520→		tflog.Info(ctx, "Successfully synced local file and updated state", map[string]interface{}{
   521→			"source": sourcePath,
   522→			"hash":   hash,
   523→			"size":   size,
   524→		})
   525→
   526→		resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
   527→		return
   528→	}
   529→
   530→	if !localFileExists {
   531→		tflog.Info(ctx, "Source file no longer exists, removing resource from state", map[string]interface{}{
   532→			"source": sourcePath,
   533→		})
   534→		// Source file was deleted - remove from state.
   535→		resp.State.RemoveResource(ctx)
   536→		return
   537→	}
   538→
   539→	// Check for remote drift if check_remote_on_plan is enabled.
   540→	hasDrift, remoteHash, err := r.checkRemoteDrift(ctx, &data)
   541→	if err != nil {
   542→		// Add warning but don't fail - drift check is informational.
   543→		resp.Diagnostics.AddWarning(
   544→			"Remote drift check failed",
   545→			fmt.Sprintf("Could not check remote file for drift: %v\n\n"+
   546→				"This may indicate connectivity issues. The apply will still check for drift.", err),
   547→		)
   548→	} else if hasDrift {
   549→		if remoteHash == "" {
   550→			resp.Diagnostics.AddWarning(
   551→				"Remote file missing",
   552→				fmt.Sprintf("Remote file %s does not exist but is expected.\n\n"+
   553→					"The file may have been deleted outside of Terraform.\n"+
   554→					"Run 'terraform apply' to recreate the file.",
   555→					data.Destination.ValueString()),
   556→			)
   557→		} else {
   558→			resp.Diagnostics.AddWarning(
   559→				"Remote file drift detected",
   560→				fmt.Sprintf("Remote file %s has been modified outside of Terraform.\n\n"+
   561→					"  Expected (from state): %s\n"+
   562→					"  Found (on remote):     %s\n\n"+
   563→					"To resolve:\n"+
   564→					"  - Run 'terraform apply' to overwrite remote with local (will fail with drift error)\n"+
   565→					"  - Run 'terraform apply -replace=%s' to force overwrite\n"+
   566→					"  - Run 'terraform import' with import_syncs_local=true to accept remote changes",
   567→					data.Destination.ValueString(),
   568→					data.SourceHash.ValueString(),
   569→					remoteHash,
   570→					fmt.Sprintf("filesync_file.%s", data.ID.ValueString())),
   571→			)
   572→		}
   573→	}
   574→
   575→	tflog.Debug(ctx, "File resource state preserved")
   576→	// State is unchanged - we preserve the existing source_hash.
   577→	resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
   578→}

   580→func (r *FileResource) Update(ctx context.Context, req resource.UpdateRequest, resp *resource.UpdateResponse) {
   581→	var data FileResourceModel
   582→	var state FileResourceModel
   583→
   584→	resp.Diagnostics.Append(req.Plan.Get(ctx, &data)...)
   585→	resp.Diagnostics.Append(req.State.Get(ctx, &state)...)
   586→	if resp.Diagnostics.HasError() {
   587→		return
   588→	}
   589→
   590→	ctx = tflog.SetField(ctx, "host", data.Host.ValueString())
   591→	ctx = tflog.SetField(ctx, "source", data.Source.ValueString())
   592→	ctx = tflog.SetField(ctx, "destination", data.Destination.ValueString())
   593→
   594→	tflog.Info(ctx, "Updating file resource")
   595→
   596→	// Calculate new source file hash.
   597→	newHash, size, err := hashFile(data.Source.ValueString())
   598→	if err != nil {
   599→		resp.Diagnostics.AddError("Failed to read source file", err.Error())
   600→		return
   601→	}
   602→	tflog.Debug(ctx, "Computed new source file hash", map[string]interface{}{
   603→		"new_hash":  newHash,
   604→		"old_hash":  state.SourceHash.ValueString(),
   605→		"size":      size,
   606→		"unchanged": newHash == state.SourceHash.ValueString(),
   607→	})
   608→
   609→	// Create SSH client.
   610→	tflog.Debug(ctx, "Establishing SSH connection")
   611→	client, err := r.createSSHClient(&data)
   612→	if err != nil {
   613→		resp.Diagnostics.AddError("Failed to create SSH connection", err.Error())
   614→		return
   615→	}
   616→	defer r.releaseSSHClient(&data, client)
   617→	tflog.Debug(ctx, "SSH connection established")
   618→
   619→	// Check for remote drift - compare remote hash with what we expect from state.
   620→	tflog.Debug(ctx, "Checking for remote drift")
   621→	remoteHash, err := client.GetFileHash(ctx, data.Destination.ValueString())
   622→	if err != nil {
   623→		// Check if file doesn't exist (could be first create after import).
   624→		// We need to distinguish between "file not found" and other errors.
   625→		exists, existsErr := client.FileExists(ctx, data.Destination.ValueString())
   626→		if existsErr != nil {
   627→			// Can't even check if file exists - report the connection/permission issue.
   628→			resp.Diagnostics.AddError(
   629→				"Failed to check remote file",
   630→				fmt.Sprintf("Could not verify remote file state at %s: %v", data.Destination.ValueString(), existsErr),
   631→			)
   632→			return
   633→		}
   634→		if exists {
   635→			// File exists but we can't read it - permission issue.
   636→			resp.Diagnostics.AddError(
   637→				"Cannot read remote file",
   638→				fmt.Sprintf("Remote file %s exists but cannot be read (permission denied?): %v", data.Destination.ValueString(), err),
   639→			)
   640→			return
   641→		}
   642→		tflog.Debug(ctx, "Remote file does not exist (first upload after import)")
   643→		// File doesn't exist - this is OK for first create after import, continue with upload.
   644→	} else if remoteHash != state.SourceHash.ValueString() {
   645→		tflog.Warn(ctx, "Remote file drift detected", map[string]interface{}{
   646→			"expected_hash": state.SourceHash.ValueString(),
   647→			"remote_hash":   remoteHash,
   648→		})
   649→		// Drift detected! Try to generate a content diff for better error message
   650→		var diffContent string
   651→
   652→		// Read local file content for diff.
   653→		localContent, localErr := os.ReadFile(data.Source.ValueString())
   654→		if localErr == nil {
   655→			// Read remote file content for diff (limit to 100KB).
   656→			remoteContent, remoteErr := client.ReadFileContent(ctx, data.Destination.ValueString(), diff.MaxDiffSize)
   657→			if remoteErr == nil {
   658→				diffContent = diff.GenerateUnifiedDiff(localContent, remoteContent, data.Destination.ValueString())
   659→			}
   660→		}
   661→
   662→		errorMsg := diff.FormatDriftError(
   663→			data.ID.ValueString(),
   664→			data.Destination.ValueString(),
   665→			state.SourceHash.ValueString(),
   666→			remoteHash,
   667→			diffContent,
   668→		)
   669→
   670→		resp.Diagnostics.AddError("Remote file drift detected", errorMsg)
   671→		return
   672→	} else {
   673→		tflog.Debug(ctx, "No remote drift detected")
   674→	}
   675→
   676→	// Upload file.
   677→	tflog.Debug(ctx, "Uploading file")
   678→	if err := client.UploadFile(ctx, data.Source.ValueString(), data.Destination.ValueString()); err != nil {
   679→		resp.Diagnostics.AddError("Failed to upload file", err.Error())
   680→		return
   681→	}
   682→	tflog.Debug(ctx, "File uploaded successfully")
   683→
   684→	// Set ownership and permissions.
   685→	tflog.Debug(ctx, "Setting file attributes", map[string]interface{}{
   686→		"owner": data.Owner.ValueString(),
   687→		"group": data.Group.ValueString(),
   688→		"mode":  data.Mode.ValueString(),
   689→	})
   690→	if err := client.SetFileAttributes(
   691→		ctx,
   692→		data.Destination.ValueString(),
   693→		data.Owner.ValueString(),
   694→		data.Group.ValueString(),
   695→		data.Mode.ValueString(),
   696→	); err != nil {
   697→		resp.Diagnostics.AddError("Failed to set file attributes", err.Error())
   698→		return
   699→	}
   700→
   701→	// Update computed values.
   702→	data.ID = types.StringValue(fmt.Sprintf("%s:%s", data.Host.ValueString(), data.Destination.ValueString()))
   703→	data.SourceHash = types.StringValue(newHash)
   704→	data.Size = types.Int64Value(size)
   705→
   706→	tflog.Info(ctx, "File resource updated successfully", map[string]interface{}{
   707→		"id":   data.ID.ValueString(),
   708→		"size": size,
   709→	})
   710→
   711→	resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
   712→}

   748→func (r *FileResource) ImportState(ctx context.Context, req resource.ImportStateRequest, resp *resource.ImportStateResponse) {
   749→	// Import format: "host:destination".
   750→	// Example: "192.168.1.100:/etc/myapp/config.json"
   751→	//
   752→	// After import, the user must update the config to set:.
   753→	// - source (required): path to the local source file
   754→	// - ssh_key_path or ssh_private_key (required): SSH credentials
   755→	//
   756→	// If import_syncs_local=true in the config, the remote file content will be
   757→	// written to the source path during the first Read after import.
   758→	//
   759→	// Then run `terraform apply` to sync state with the config.
   760→
   761→	id := req.ID
   762→	tflog.Info(ctx, "Importing file resource", map[string]interface{}{
   763→		"import_id": id,
   764→	})
   765→
   766→	// Parse the import ID - format is "host:destination".
   767→	// The destination is an absolute path, so we split on the first ":".
   768→	colonIdx := strings.Index(id, ":")
   769→	if colonIdx == -1 || colonIdx == 0 || colonIdx == len(id)-1 {
   770→		resp.Diagnostics.AddError(
   771→			"Invalid Import ID",
   772→			fmt.Sprintf(
   773→				"Import ID must be in format 'host:destination' (e.g., '192.168.1.100:/etc/myapp/config.json').\n"+
   774→					"Got: %s", id,
   775→			),
   776→		)
   777→		return
   778→	}
   779→
   780→	host := id[:colonIdx]
   781→	destination := id[colonIdx+1:]
   782→
   783→	// Validate destination is an absolute path.
   784→	if !strings.HasPrefix(destination, "/") {
   785→		resp.Diagnostics.AddError(
   786→			"Invalid Import ID",
   787→			fmt.Sprintf(
   788→				"Destination must be an absolute path starting with '/'.\n"+
   789→					"Got: %s", destination,
   790→			),
   791→		)
   792→		return
   793→	}
   794→
   795→	resp.Diagnostics.Append(resp.State.SetAttribute(ctx, path.Root("id"), id)...)
   796→	resp.Diagnostics.Append(resp.State.SetAttribute(ctx, path.Root("host"), host)...)
   797→	resp.Diagnostics.Append(resp.State.SetAttribute(ctx, path.Root("destination"), destination)...)
   798→}

   800→// syncLocalFromRemote downloads the remote file content to the local source path.
   801→// This is called during Read when import_syncs_local=true and the local file
   802→// doesn't exist or differs from remote.
   803→func (r *FileResource) syncLocalFromRemote(ctx context.Context, data *FileResourceModel) error {
   804→	sourcePath := data.Source.ValueString()
   805→	if sourcePath == "" {
   806→		return fmt.Errorf("source path is not set - add 'source' attribute to your config")
   807→	}
   808→
   809→	tflog.Info(ctx, "Syncing local file from remote (import_syncs_local=true)", map[string]interface{}{
   810→		"source":      sourcePath,
   811→		"destination": data.Destination.ValueString(),
   812→	})
   813→
   814→	// Create SSH client.
   815→	client, err := r.createSSHClient(data)
   816→	if err != nil {
   817→		return fmt.Errorf("failed to create SSH connection: %w", err)
   818→	}
   819→	defer r.releaseSSHClient(data, client)
   820→
   821→	// Check if remote file exists.
   822→	exists, err := client.FileExists(ctx, data.Destination.ValueString())
   823→	if err != nil {
   824→		return fmt.Errorf("failed to check if remote file exists: %w", err)
   825→	}
   826→	if !exists {
   827→		return fmt.Errorf("remote file %s does not exist", data.Destination.ValueString())
   828→	}
   829→
   830→	// Read remote file content.
   831→	content, err := client.ReadFileContent(ctx, data.Destination.ValueString(), 0) // 0 = no limit
   832→	if err != nil {
   833→		return fmt.Errorf("failed to read remote file content: %w", err)
   834→	}
   835→
   836→	// Create parent directories if needed.
   837→	dir := filepath.Dir(sourcePath)
   838→	if err := os.MkdirAll(dir, 0755); err != nil {
   839→		return fmt.Errorf("failed to create parent directory %s: %w", dir, err)
   840→	}
   841→
   842→	// Write content to local file atomically (write to temp, then rename).
   843→	tmpPath := sourcePath + ".tmp"
   844→	if err := os.WriteFile(tmpPath, content, 0644); err != nil {
   845→		return fmt.Errorf("failed to write temporary file %s: %w", tmpPath, err)
   846→	}
   847→
   848→	if err := os.Rename(tmpPath, sourcePath); err != nil {
   849→		os.Remove(tmpPath) // Clean up temp file on error.
   850→		return fmt.Errorf("failed to rename temporary file to %s: %w", sourcePath, err)
   851→	}
   852→
   853→	tflog.Info(ctx, "Successfully synced local file from remote", map[string]interface{}{
   854→		"source": sourcePath,
   855→		"size":   len(content),
   856→	})
   857→
   858→	return nil
   859→}

   875→// createSSHClient creates or retrieves an SSH client (from pool if enabled).
   876→func (r *FileResource) createSSHClient(data *FileResourceModel) (gosftp.ClientInterface, error) {
   877→	config := r.getSSHConfig(data)
   878→
   879→	// Use connection pool if enabled.
   880→	if r.isPoolingEnabled() && r.providerConfig != nil && r.providerConfig.pool != nil {
   881→		return r.providerConfig.pool.GetOrCreate(config)
   882→	}
   883→
   884→	// Otherwise, create a new connection using the factory.
   885→	factory := r.sshClientFactory
   886→	if factory == nil {
   887→		factory = DefaultSSHClientFactory
   888→	}
   889→	client, err := factory(config)
   890→	if err != nil {
   891→		// Provide detailed error message for debugging SSH connection issues.
   892→		return nil, fmt.Errorf("%w (host=%s, port=%d, user=%s, insecure_ignore_host_key=%v, has_key=%v, has_password=%v)",
   893→			err,
   894→			config.Host,
   895→			config.Port,
   896→			config.User,
   897→			config.InsecureIgnoreHostKey,
   898→			config.KeyPath != "" || config.PrivateKey != "",
   899→			config.Password != "",
   900→		)
   901→	}
   902→	return client, nil
   903→}

   905→// releaseSSHClient releases a connection back to the pool (if pooling enabled).

   912→	} else {
   913→		// Not pooling - close the connection.
   914→		client.Close()
   915→	}

   918→func hashFile(path string) (string, int64, error) {
   919→	f, err := os.Open(path)
   920→	if err != nil {
   921→		return "", 0, err
   922→	}
   923→	defer f.Close()
   924→
   925→	h := sha256.New()
   926→	size, err := io.Copy(h, f)
   927→	if err != nil {
   928→		return "", 0, err
   929→	}
   930→
   931→	return "sha256:" + hex.EncodeToString(h.Sum(nil)), size, nil
   932→}

   974→// remoteDriftChecker is used to check for remote drift during the Read phase.
   975→// This is called when check_remote_on_plan is true.
   976→// Note: We can't check during plan modifiers because they don't have access to
   977→// provider config (SSH credentials). Instead, we check during Read which runs
   978→// as part of refresh during plan.
   979→func (r *FileResource) checkRemoteDrift(ctx context.Context, data *FileResourceModel) (hasDrift bool, remoteHash string, err error) {
   980→	// Skip if check_remote_on_plan is not enabled.
   981→	if data.CheckRemoteOnPlan.IsNull() || !data.CheckRemoteOnPlan.ValueBool() {
   982→		return false, "", nil
   983→	}
   984→
   985→	// Skip if we don't have state hash to compare against.
   986→	if data.SourceHash.IsNull() || data.SourceHash.IsUnknown() {
   987→		return false, "", nil
   988→	}
   989→
   990→	tflog.Debug(ctx, "Checking remote for drift (check_remote_on_plan=true)")
   991→
   992→	// Create SSH client.
   993→	client, err := r.createSSHClient(data)
   994→	if err != nil {
   995→		return false, "", fmt.Errorf("failed to create SSH connection for drift check: %w", err)
   996→	}
   997→	defer r.releaseSSHClient(data, client)
   998→
   999→	// Check if file exists.
  1000→	exists, err := client.FileExists(ctx, data.Destination.ValueString())
  1001→	if err != nil {
  1002→		return false, "", fmt.Errorf("failed to check if remote file exists: %w", err)
  1003→	}
  1004→	if !exists {
  1005→		// File doesn't exist on remote - this is drift (file was deleted).
  1006→		return true, "", nil
  1007→	}
  1008→
  1009→	// Get remote file hash.
  1010→	remoteHash, err = client.GetFileHash(ctx, data.Destination.ValueString())
  1011→	if err != nil {
  1012→		return false, "", fmt.Errorf("failed to get remote file hash: %w", err)
  1013→	}
  1014→
  1015→	// Compare with state hash.
  1016→	stateHash := data.SourceHash.ValueString()
  1017→	if remoteHash != stateHash {
  1018→		tflog.Warn(ctx, "Remote drift detected", map[string]interface{}{
  1019→			"state_hash":  stateHash,
  1020→			"remote_hash": remoteHash,
  1021→		})
  1022→		return true, remoteHash, nil
  1023→	}
  1024→
  1025→	tflog.Debug(ctx, "No remote drift detected")
  1026→	return false, remoteHash, nil
  1027→}
