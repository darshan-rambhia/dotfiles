   191→// TestFileResource_InsecureIgnoreHostKey tests that the insecure_ignore_host_key setting
   192→// is properly passed through to the SSH client config at resource and provider levels.
   193→func TestFileResource_InsecureIgnoreHostKey(t *testing.T) {
   194→	tests := []struct {
   195→		name                      string
   196→		providerConfig            *FilesyncProviderModel
   197→		data                      *FileResourceModel
   198→		wantInsecureIgnoreHostKey bool
   199→	}{
   200→		{
   201→			name: "resource level insecure_ignore_host_key = true",
   202→			data: &FileResourceModel{
   203→				Host:                  types.StringValue("192.168.1.100"),
   204→				SSHPort:               types.Int64Value(22),
   205→				SSHUser:               types.StringValue("root"),
   206→				SSHKeyPath:            types.StringValue("~/.ssh/id_ed25519"),
   207→				InsecureIgnoreHostKey: types.BoolValue(true),
   208→			},
   209→			wantInsecureIgnoreHostKey: true,
   210→		},
   211→		{
   212→			name: "resource level insecure_ignore_host_key = false (explicit)",
   213→			data: &FileResourceModel{
   214→				Host:                  types.StringValue("192.168.1.100"),
   215→				SSHPort:               types.Int64Value(22),
   216→				SSHUser:               types.StringValue("root"),
   217→				SSHKeyPath:            types.StringValue("~/.ssh/id_ed25519"),
   218→				InsecureIgnoreHostKey: types.BoolValue(false),
   219→			},
   220→			wantInsecureIgnoreHostKey: false,
   221→		},
   222→		{
   223→			name: "resource level insecure_ignore_host_key not set (null)",
   224→			data: &FileResourceModel{
   225→				Host:                  types.StringValue("192.168.1.100"),
   226→				SSHPort:               types.Int64Value(22),
   227→				SSHUser:               types.StringValue("root"),
   228→				SSHKeyPath:            types.StringValue("~/.ssh/id_ed25519"),
   229→				InsecureIgnoreHostKey: types.BoolNull(),
   230→			},
   231→			wantInsecureIgnoreHostKey: false,
   232→		},
   233→		{
   234→			name: "provider level insecure_ignore_host_key = true, resource not set",
   235→			providerConfig: &FilesyncProviderModel{
   236→				SSHKeyPath:            types.StringValue("~/.ssh/id_ed25519"),
   237→				InsecureIgnoreHostKey: types.BoolValue(true),
   238→			},
   239→			data: &FileResourceModel{
   240→				Host:                  types.StringValue("192.168.1.100"),
   241→				SSHPort:               types.Int64Value(22),
   242→				SSHUser:               types.StringValue("root"),
   243→				InsecureIgnoreHostKey: types.BoolNull(), // Not set at resource level
   244→			},
   245→			wantInsecureIgnoreHostKey: true,
   246→		},
   247→		{
   248→			name: "resource level overrides provider level (resource=true, provider=false)",
   249→			providerConfig: &FilesyncProviderModel{
   250→				SSHKeyPath:            types.StringValue("~/.ssh/id_ed25519"),
   251→				InsecureIgnoreHostKey: types.BoolValue(false),
   252→			},
   253→			data: &FileResourceModel{
   254→				Host:                  types.StringValue("192.168.1.100"),
   255→				SSHPort:               types.Int64Value(22),
   256→				SSHUser:               types.StringValue("root"),
   257→				InsecureIgnoreHostKey: types.BoolValue(true),
   258→			},
   259→			wantInsecureIgnoreHostKey: true,
   260→		},
   261→		{
   262→			name: "resource level overrides provider level (resource=false, provider=true)",
   263→			providerConfig: &FilesyncProviderModel{
   264→				SSHKeyPath:            types.StringValue("~/.ssh/id_ed25519"),
   265→				InsecureIgnoreHostKey: types.BoolValue(true),
   266→			},
   267→			data: &FileResourceModel{
   268→				Host:                  types.StringValue("192.168.1.100"),
   269→				SSHPort:               types.Int64Value(22),
   270→				SSHUser:               types.StringValue("root"),
   271→				InsecureIgnoreHostKey: types.BoolValue(false),
   272→			},
   273→			// Resource explicitly sets false, which should override provider's true.
   274→			// This ensures users can enable host key checking at the resource level
   275→			// even if the provider has it disabled globally.
   276→			wantInsecureIgnoreHostKey: false,
   277→		},
   278→	}
   279→
   280→	for _, tt := range tests {
   281→		t.Run(tt.name, func(t *testing.T) {
   282→			factory := &ConfigCapturingFactory{
   283→				Mock: NewMockSSHClient(),
   284→			}
   285→			r := &FileResource{
   286→				providerConfig:   tt.providerConfig,
   287→				sshClientFactory: factory.Factory(),
   288→			}
   289→
   290→			_, err := r.createSSHClient(tt.data)
   291→			if err != nil {
   292→				t.Fatalf("createSSHClient() unexpected error: %v", err)
   293→			}
   294→
   295→			if factory.CapturedConfig.InsecureIgnoreHostKey != tt.wantInsecureIgnoreHostKey {
   296→				t.Errorf("InsecureIgnoreHostKey = %v, want %v",
   297→					factory.CapturedConfig.InsecureIgnoreHostKey, tt.wantInsecureIgnoreHostKey)
   298→			}
   299→		})
   300→	}
   301→}

   347→		},
   348→		{
   349→			name:        "relative path",

   352→		},
   353→	}

   355→	for _, tt := range tests {

   358→

   360→			var schemaResp resource.SchemaResponse
   361→			r.Schema(context.Background(), resource.SchemaRequest{}, &schemaResp)
   362→
   363→			state := tfsdk.State{
   364→				Schema: schemaResp.Schema,
   365→				Raw:    tftypes.NewValue(schemaResp.Schema.Type().TerraformType(context.Background()), nil),

   369→				State: state,

   394→				t.Errorf("host = %q, want %q", host.ValueString(), tt.wantHost)

   469→// TestHashFileUnit tests the hashFile function with various file types.
   470→func TestHashFileUnit(t *testing.T) {
   471→	tests := []struct {
   472→		name     string
   473→		filename string
   474→		content  []byte
   475→	}{
   476→		{
   477→			name:     "binary content",
   478→			filename: "binary.bin",
   479→			content:  []byte{0x00, 0x01, 0x02, 0xff, 0xfe},
   480→		},
   481→		{
   482→			name:     "unicode content",
   483→			filename: "unicode.txt",
   484→			content:  []byte("héllo wörld 你好世界 🌍"),
   485→		},
   486→		{
   487→			name:     "empty file",
   488→			filename: "empty.txt",
   489→			content:  []byte{},
   490→		},
   491→		{
   492→			name:     "large content",
   493→			filename: "large.txt",
   494→			content:  make([]byte, 1024*10), // 10KB
   495→		},
   496→		{
   497→			name:     "newlines only",
   498→			filename: "newlines.txt",
   499→			content:  []byte("\n\n\n\n\n"),
   500→		},
   501→	}
   502→
   503→	for _, tt := range tests {
   504→		t.Run(tt.name, func(t *testing.T) {
   505→			tmpDir := t.TempDir()
   506→			filePath := filepath.Join(tmpDir, tt.filename)
   507→
   508→			if err := os.WriteFile(filePath, tt.content, 0644); err != nil {
   509→				t.Fatalf("failed to write file: %v", err)
   510→			}
   511→
   512→			hash, size, err := hashFile(filePath)
   513→			if err != nil {
   514→				t.Fatalf("hashFile() error: %v", err)
   515→			}
   516→
   517→			if size != int64(len(tt.content)) {
   518→				t.Errorf("hashFile() size = %d, want %d", size, len(tt.content))
   519→			}
   520→
   521→			if len(hash) < 7 || hash[:7] != "sha256:" {
   522→				t.Error("hashFile() hash should have sha256: prefix")
   523→			}
   524→		})
   525→	}
   526→}
