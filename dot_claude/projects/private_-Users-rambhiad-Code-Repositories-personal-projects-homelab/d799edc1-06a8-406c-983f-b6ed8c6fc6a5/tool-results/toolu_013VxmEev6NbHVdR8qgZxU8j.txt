     1→# =============================================================================
     2→# Docker LXC Module
     3→# =============================================================================
     4→# Opinionated module for deploying Docker-based applications in LXC containers.
     5→# Uses lxc-base + filesync provider + remotedockercompose provider for a
     6→# fully declarative deployment experience.
     7→# =============================================================================
     8→
     9→terraform {
    10→  required_providers {
    11→    proxmox = {
    12→      source  = "bpg/proxmox"
    13→      version = "~> 0.69"
    14→    }
    15→    filesync = {
    16→      source  = "darshan-rambhia/filesync"
    17→      version = "~> 0.3"
    18→    }
    19→    ansible = {
    20→      source  = "marshallford/ansible"
    21→      version = "~> 0.9"
    22→    }
    23→  }
    24→}
    25→
    26→# -----------------------------------------------------------------------------
    27→# LXC Container
    28→# -----------------------------------------------------------------------------
    29→
    30→module "lxc" {
    31→  source = "../lxc-base"
    32→
    33→  name         = var.name
    34→  container_id = var.container_id
    35→  proxmox_node = var.proxmox_node
    36→
    37→  # Network - dual-network architecture
    38→  container_ip      = var.container_ip
    39→  container_gateway = var.container_gateway
    40→  lan_ip            = var.lan_ip
    41→  lan_gateway       = var.lan_gateway
    42→
    43→  # Resources
    44→  cores  = var.cores
    45→  memory = var.memory
    46→  swap   = var.swap
    47→
    48→  rootfs_storage = var.rootfs_storage
    49→  rootfs_size    = var.rootfs_size
    50→
    51→  # Container type
    52→  unprivileged = var.unprivileged
    53→  template     = var.template
    54→
    55→  # SSH
    56→  ssh_public_key               = var.ssh_public_key
    57→  proxmox_ssh_host             = var.proxmox_ssh_host
    58→  proxmox_ssh_private_key_path = var.ssh_private_key_path
    59→
    60→  # Features
    61→  enable_nesting = true
    62→  enable_keyctl  = true
    63→  enable_fuse    = false
    64→
    65→  # GPU passthrough (AMD Radeon - /dev/kfd, /dev/dri)
    66→  enable_gpu = var.enable_gpu
    67→
    68→  # Mount points
    69→  mount_points = var.mount_points
    70→
    71→  # Proxmox notes
    72→  description = var.enable_proxmox_notes ? templatefile("${path.module}/templates/notes.md.tftpl", {
    73→    service_name = var.service_name != "" ? var.service_name : var.name
    74→    description  = var.service_description
    75→    urls         = var.service_urls
    76→    services     = var.services
    77→    ssh_host     = split("/", var.container_ip)[0]
    78→    custom_notes = var.custom_notes
    79→    managed_by   = true
    80→  }) : ""
    81→
    82→  # Tags
    83→  tags = concat(["docker"], var.tags)
    84→}
    85→
    86→# -----------------------------------------------------------------------------
    87→# File Sync - Using filesync provider directly
    88→# -----------------------------------------------------------------------------
    89→
    90→# Get list of all files in deploy directory, excluding specified patterns
    91→locals {
    92→  _all_deploy_files = var.deploy_dir != "" ? fileset(var.deploy_dir, "**") : []
    93→
    94→  # Filter out files matching exclude patterns
    95→  deploy_files = [
    96→    for f in local._all_deploy_files : f
    97→    if !anytrue([for pattern in var.exclude_patterns : can(regex(pattern, f))])
    98→  ]
    99→}
   100→
   101→# Sync each file individually using filesync provider
   102→resource "filesync_file" "config_files" {
   103→  for_each = toset(local.deploy_files)
   104→
   105→  source      = "${var.deploy_dir}/${each.value}"
   106→  destination = "${var.target_path}/${each.value}"
   107→  host        = module.lxc.container_ip
   108→  mode        = "0644"
   109→
   110→  # SSH configuration - provider config is inherited but explicit for clarity
   111→  ssh_user     = "root"
   112→  ssh_key_path = var.ssh_private_key_path
   113→
   114→  depends_on = [module.lxc]
   115→}
   116→
   117→# Set custom permissions for specific files
   118→resource "filesync_file" "custom_permissions" {
   119→  for_each = var.file_permissions
   120→
   121→  source      = "${var.deploy_dir}/${trimsuffix(trimprefix(each.key, var.target_path), "/")}"
   122→  destination = each.key
   123→  host        = module.lxc.container_ip
   124→  mode        = each.value
   125→
   126→  # SSH configuration
   127→  ssh_user     = "root"
   128→  ssh_key_path = var.ssh_private_key_path
   129→
   130→  depends_on = [filesync_file.config_files]
   131→}
   132→
   133→# -----------------------------------------------------------------------------
   134→# Docker Compose Deployment via Ansible
   135→# -----------------------------------------------------------------------------
   136→# After filesync completes, Ansible deploys and starts the containers
   137→
   138→resource "ansible_navigator_run" "deploy_containers" {
   139→  depends_on = [
   140→    filesync_file.config_files,
   141→    filesync_file.custom_permissions
   142→  ]
   143→
   144→  triggers = {
   145→    # Use 'run' key to trigger playbook re-run when config files change
   146→    # See: https://registry.terraform.io/providers/marshallford/ansible/latest/docs/resources/navigator_run
   147→    run = sha256(join(",", [for f in local.deploy_files : filesha256("${var.deploy_dir}/${f}")]))
   148→  }
   149→
   150→  execution_environment = {
   151→    enabled = false  # Run ansible directly on host
   152→  }
   153→
   154→  ansible_options = {
   155→    private_keys = [{
   156→      name = "homelab"
   157→      data = file(var.ssh_private_key_path)
   158→    }]
   159→  }
   160→
   161→  playbook = <<-EOT
   162→    - name: Deploy Docker Compose for ${var.name}
   163→      hosts: lxc
   164→      gather_facts: no
   165→      vars:
   166→        config_dir: ${var.target_path}
   167→        app_name: ${var.name}
   168→
   169→      tasks:
   170→        # =====================================================================
   171→        # Verify Required Files
   172→        # =====================================================================
   173→        - name: Check docker-compose.yml exists
   174→          ansible.builtin.stat:
   175→            path: "{{ config_dir }}/docker-compose.yml"
   176→          register: compose_file
   177→
   178→        - name: Fail if docker-compose.yml missing
   179→          ansible.builtin.fail:
   180→            msg: "Required file missing: docker-compose.yml"
   181→          when: not compose_file.stat.exists
   182→
   183→        # =====================================================================
   184→        # Pull Images & Start Containers
   185→        # =====================================================================
   186→        - name: Pull container images
   187→          ansible.builtin.command:
   188→            cmd: docker compose -f {{ config_dir }}/docker-compose.yml pull
   189→            chdir: "{{ config_dir }}"
   190→          register: pull_result
   191→          changed_when: "'Pulled' in pull_result.stdout or 'Downloaded' in pull_result.stdout"
   192→
   193→        - name: Start containers
   194→          ansible.builtin.command:
   195→            cmd: docker compose -f {{ config_dir }}/docker-compose.yml up -d --remove-orphans
   196→            chdir: "{{ config_dir }}"
   197→          register: up_result
   198→          changed_when: "'Started' in up_result.stderr or 'Creating' in up_result.stderr or 'Created' in up_result.stderr"
   199→
   200→        # =====================================================================
   201→        # Health Checks
   202→        # =====================================================================
   203→        - name: Wait for containers to stabilize
   204→          ansible.builtin.pause:
   205→            seconds: 10
   206→
   207→        - name: Get container status
   208→          ansible.builtin.command:
   209→            cmd: docker compose -f {{ config_dir }}/docker-compose.yml ps --format json
   210→            chdir: "{{ config_dir }}"
   211→          register: ps_result
   212→          changed_when: false
   213→
   214→        - name: Verify all containers are running
   215→          ansible.builtin.assert:
   216→            that:
   217→              - ps_result.stdout | length > 0
   218→            fail_msg: "No containers found for {{ app_name }}"
   219→            success_msg: "Containers deployed successfully for {{ app_name }}"
   220→  EOT
   221→
   222→  inventory = yamlencode({
   223→    lxc = {
   224→      hosts = {
   225→        "${var.name}" = {
   226→          ansible_host = module.lxc.container_ip
   227→          ansible_user = "root"
   228→        }
   229→      }
   230→    }
   231→  })
   232→}
