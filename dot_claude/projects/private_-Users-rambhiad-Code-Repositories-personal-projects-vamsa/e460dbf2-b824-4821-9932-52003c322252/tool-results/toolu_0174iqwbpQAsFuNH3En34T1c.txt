     1→#!/usr/bin/env bun
     2→/**
     3→ * Database management script
     4→ *
     5→ * Usage:
     6→ *   bun run db start      # Start PostgreSQL container
     7→ *   bun run db stop       # Stop PostgreSQL container
     8→ *   bun run db migrate    # Run Prisma migrations
     9→ *   bun run db push       # Push schema to database
    10→ *   bun run db seed       # Seed the database
    11→ *   bun run db generate   # Generate Prisma client
    12→ *   bun run db studio     # Open Prisma Studio
    13→ *   bun run db reset      # Reset database (destructive)
    14→ */
    15→
    16→import { $ } from "bun";
    17→import { existsSync } from "fs";
    18→import { resolve } from "path";
    19→
    20→const command = process.argv[2];
    21→
    22→const runCommand = async (label: string, cmd: string[]) => {
    23→  console.log(`\n📋 ${label}...`);
    24→  try {
    25→    await $`${cmd}`;
    26→    console.log(`✅ ${label} complete`);
    27→  } catch (error) {
    28→    console.error(`❌ ${label} failed`);
    29→    process.exit(1);
    30→  }
    31→};
    32→
    33→const projectRoot = resolve(import.meta.dir, "..");
    34→const dockerComposePath = resolve(projectRoot, "docker/docker-compose.dev.yml");
    35→
    36→const commands: Record<string, () => Promise<void>> = {
    37→  start: async () => {
    38→    console.log("🚀 Starting PostgreSQL...");
    39→    // Check if already running
    40→    try {
    41→      await $`docker compose -f ${dockerComposePath} ps --status running db`;
    42→      console.log("✅ PostgreSQL is already running");
    43→      return;
    44→    } catch {
    45→      // Not running, start it
    46→      await runCommand(
    47→        "Starting PostgreSQL container",
    48→        ["docker", "compose", "-f", dockerComposePath, "up", "-d", "--wait"]
    49→      );
    50→    }
    51→  },
    52→
    53→  stop: async () => {
    54→    await runCommand(
    55→      "Stopping PostgreSQL",
    56→      ["docker", "compose", "-f", dockerComposePath, "down"]
    57→    );
    58→  },
    59→
    60→  migrate: async () => {
    61→    await runCommand(
    62→      "Running migrations",
    63→      ["bunx", "prisma", "migrate", "dev"]
    64→    );
    65→  },
    66→
    67→  push: async () => {
    68→    await runCommand(
    69→      "Pushing schema to database",
    70→      ["bunx", "prisma", "db", "push"]
    71→    );
    72→  },
    73→
    74→  seed: async () => {
    75→    await runCommand("Seeding database", ["bun", "run", "prisma/seed.ts"]);
    76→  },
    77→
    78→  generate: async () => {
    79→    await runCommand("Generating Prisma client", ["bunx", "prisma", "generate"]);
    80→  },
    81→
    82→  studio: async () => {
    83→    console.log("🎨 Opening Prisma Studio...");
    84→    await $`bunx prisma studio`;
    85→  },
    86→
    87→  reset: async () => {
    88→    console.log("⚠️  Warning: This will delete all data in the database!");
    89→    const confirm = process.argv[3];
    90→    if (confirm !== "--force") {
    91→      console.log("Run with --force to confirm");
    92→      return;
    93→    }
    94→    await runCommand(
    95→      "Resetting database",
    96→      ["bunx", "prisma", "migrate", "reset", "--force"]
    97→    );
    98→  },
    99→};
   100→
   101→const printUsage = () => {
   102→  console.log(`
   103→Database Management Commands
   104→
   105→Usage: bun run db <command> [options]
   106→
   107→Commands:
   108→  start      Start PostgreSQL container
   109→  stop       Stop PostgreSQL container
   110→  migrate    Run pending migrations
   111→  push       Push schema changes to database
   112→  seed       Seed the database with initial data
   113→  generate   Generate Prisma client
   114→  studio     Open Prisma Studio
   115→  reset      Reset database (destructive) - requires --force flag
   116→
   117→Examples:
   118→  bun run db start
   119→  bun run db migrate
   120→  bun run db reset --force
   121→  `);
   122→};
   123→
   124→const main = async () => {
   125→  if (!command || !commands[command]) {
   126→    printUsage();
   127→    process.exit(command ? 1 : 0);
   128→  }
   129→
   130→  try {
   131→    await commands[command]();
   132→  } catch (error) {
   133→    console.error("Error:", error);
   134→    process.exit(1);
   135→  }
   136→};
   137→
   138→main();
