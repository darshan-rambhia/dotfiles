     1→#!/usr/bin/env bun
     2→/**
     3→ * Validation script - runs all checks: lint, typecheck, test, format
     4→ *
     5→ * Usage:
     6→ *   bun run validate              # Run all checks sequentially
     7→ *   bun run validate --parallel   # Run all checks in parallel
     8→ *   bun run validate --check      # Dry-run (no fixes)
     9→ */
    10→
    11→import { $ } from "bun";
    12→
    13→const args = process.argv.slice(2);
    14→const isParallel = args.includes("--parallel");
    15→const isCheck = args.includes("--check");
    16→
    17→const commands = {
    18→  test: ["bun", "test"],
    19→  lint: isCheck ? ["eslint", "."] : ["eslint", ".", "--fix"],
    20→  typecheck: ["tsc"],
    21→  format: isCheck
    22→    ? ["prettier", "--check", "."]
    23→    : ["prettier", "--write", "."],
    24→};
    25→
    26→const runCommand = async (label: string, cmd: string[]) => {
    27→  console.log(`\n📋 ${label}...`);
    28→  try {
    29→    await $`${cmd}`;
    30→    console.log(`✅ ${label} passed`);
    31→    return true;
    32→  } catch (error) {
    33→    console.error(`❌ ${label} failed`);
    34→    return false;
    35→  }
    36→};
    37→
    38→const runParallel = async () => {
    39→  console.log("🚀 Running validation checks in parallel...\n");
    40→
    41→  const results = await Promise.allSettled([
    42→    runCommand("Tests", commands.test),
    43→    runCommand("Linting", commands.lint),
    44→    runCommand("Type checking", commands.typecheck),
    45→    runCommand("Formatting", commands.format),
    46→  ]);
    47→
    48→  const failures = results.filter((r) => r.status === "rejected" || !r.value);
    49→  if (failures.length > 0) {
    50→    console.error("\n❌ Some checks failed");
    51→    process.exit(1);
    52→  }
    53→
    54→  console.log("\n✅ All validation checks passed!");
    55→};
    56→
    57→const runSequential = async () => {
    58→  console.log("🚀 Running validation checks sequentially...\n");
    59→
    60→  const checks = [
    61→    ["Tests", commands.test],
    62→    ["Linting", commands.lint],
    63→    ["Type checking", commands.typecheck],
    64→    ["Formatting", commands.format],
    65→  ] as const;
    66→
    67→  for (const [label, cmd] of checks) {
    68→    const success = await runCommand(label, cmd);
    69→    if (!success) {
    70→      console.error(`\n❌ Validation stopped at ${label}`);
    71→      process.exit(1);
    72→    }
    73→  }
    74→
    75→  console.log("\n✅ All validation checks passed!");
    76→};
    77→
    78→const main = async () => {
    79→  if (isCheck) {
    80→    console.log("🔍 Running in CHECK mode (no fixes applied)\n");
    81→  }
    82→
    83→  if (isParallel) {
    84→    await runParallel();
    85→  } else {
    86→    await runSequential();
    87→  }
    88→};
    89→
    90→main().catch((err) => {
    91→  console.error("Fatal error:", err);
    92→  process.exit(1);
    93→});
