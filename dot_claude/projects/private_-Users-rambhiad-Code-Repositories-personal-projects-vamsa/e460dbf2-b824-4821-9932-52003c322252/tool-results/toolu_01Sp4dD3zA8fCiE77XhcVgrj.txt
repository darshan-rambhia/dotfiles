    13→// Load environment variables from .env file

    23→        if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) {
    24→          value = value.slice(1, -1);
    25→        }
    26→        if (!process.env[key]) {
    27→          process.env[key] = value;
    28→        }
    29→      }
    30→    });
    31→  } catch {
    32→    // .env file not found, use defaults
