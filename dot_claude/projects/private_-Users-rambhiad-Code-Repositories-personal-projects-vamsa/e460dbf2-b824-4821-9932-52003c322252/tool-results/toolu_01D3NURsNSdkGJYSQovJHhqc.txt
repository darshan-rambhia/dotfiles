     1→#!/usr/bin/env bun

     3→ * Docker management script for production deployments

     5→ * Usage:
     6→ *   bun run docker build [--sqlite]       # Build Docker image

    12→import { $ } from "bun";
    13→import { resolve } from "path";

    15→const command = process.argv[2];
    16→const isSqlite = process.argv.includes("--sqlite");
    17→
    18→const projectRoot = resolve(import.meta.dir, "..");
    19→
    20→const shellScript = (name: string) => {
    21→  const suffix = isSqlite ? "sqlite" : "";
    22→  return resolve(projectRoot, `scripts/${name}${suffix ? ".sqlite" : ""}.sh`);
    23→};
    24→
    25→const runCommand = async (label: string, cmd: string[]) => {
    26→  console.log(`\n📋 ${label}...`);
    27→  try {
    28→    await $`${cmd}`;
    29→    console.log(`✅ ${label} complete`);
    30→  } catch (error) {
    31→    console.error(`❌ ${label} failed`);
    32→    process.exit(1);
    33→  }
    34→};
    35→
    36→const commands: Record<string, () => Promise<void>> = {
    37→  build: async () => {
    38→    const script = shellScript("build");
    39→    const args = isSqlite ? ["sqlite"] : [];
    40→    console.log(`🐳 Building Docker image${isSqlite ? " (SQLite)" : ""}...`);
    41→    await runCommand("Docker build", ["bash", script, ...args]);
    42→  },
    43→
    44→  run: async () => {
    45→    const script = shellScript("run");
    46→    const args = isSqlite ? ["sqlite"] : [];

    49→  },

    51→  up: async () => {
    52→    const script = shellScript("up");
    53→    const args = isSqlite ? ["sqlite"] : [];
    54→    console.log(
    55→      `🐳 Starting docker-compose${isSqlite ? " (SQLite)" : ""}...`
    56→    );
    57→    await runCommand("Docker compose up", ["bash", script, ...args]);
    58→  },
    59→
    60→  down: async () => {
    61→    const script = shellScript("down");
    62→    const args = isSqlite ? ["sqlite"] : [];
    63→    console.log(
    64→      `🐳 Stopping docker-compose${isSqlite ? " (SQLite)" : ""}...`
    65→    );
    66→    await runCommand("Docker compose down", ["bash", script, ...args]);
    67→  },
    68→};
    69→
    70→const printUsage = () => {
    71→  console.log(`
    72→Docker Management Commands
    73→
    74→Usage: bun run docker <command> [options]
    75→
    76→Commands:
    77→  build      Build Docker image
    78→  run        Run Docker container
    79→  up         Start services via docker-compose
    80→  down       Stop services via docker-compose
    81→
    82→Options:
    83→  --sqlite   Use SQLite variant (default: PostgreSQL)
    84→
    85→Examples:
    86→  bun run docker build

    88→  bun run docker up

    90→  bun run docker down --sqlite
    91→  `);
    92→};
    93→
    94→const main = async () => {
    95→  if (!command || !commands[command]) {
    96→    printUsage();
    97→    process.exit(command ? 1 : 0);
    98→  }
