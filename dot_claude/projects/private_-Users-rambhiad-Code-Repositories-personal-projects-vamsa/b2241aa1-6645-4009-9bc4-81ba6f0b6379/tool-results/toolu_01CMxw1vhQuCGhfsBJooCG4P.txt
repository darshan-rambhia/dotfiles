     1→import unzipper from "unzipper";
     2→import { z } from "zod";
     3→import { db } from "@/lib/db";
     4→import {
     5→  backupMetadataSchema,
     6→  type BackupMetadata,
     7→  type Conflict,
     8→  type ValidationResult,
     9→} from "@/schemas/backup";
    10→
    11→// Supported backup versions
    12→const SUPPORTED_VERSIONS = ["1.0.0"];
    13→const CURRENT_VERSION = "1.0.0";
    14→
    15→// Data schemas for validation
    16→const personDataSchema = z.object({
    17→  id: z.string(),
    18→  firstName: z.string(),
    19→  lastName: z.string(),
    20→  email: z.string().nullable().optional(),
    21→  createdAt: z.string(),
    22→  updatedAt: z.string(),
    23→});
    24→
    25→const userDataSchema = z.object({
    26→  id: z.string(),
    27→  email: z.string(),
    28→  name: z.string().nullable().optional(),
    29→  personId: z.string().nullable().optional(),
    30→  role: z.enum(["ADMIN", "MEMBER", "VIEWER"]),
    31→  createdAt: z.string(),
    32→  updatedAt: z.string(),
    33→});
    34→
    35→const relationshipDataSchema = z.object({
    36→  id: z.string(),
    37→  personId: z.string(),
    38→  relatedPersonId: z.string(),
    39→  type: z.enum(["PARENT", "CHILD", "SPOUSE", "SIBLING"]),
    40→  createdAt: z.string(),
    41→  updatedAt: z.string(),
    42→});
    43→
    44→export class BackupValidator {
    45→  private zipBuffer: Buffer;
    46→  private extractedFiles: Map<string, any> = new Map();
    47→  private metadata: BackupMetadata | null = null;
    48→
    49→  constructor(zipBuffer: Buffer) {
    50→    this.zipBuffer = zipBuffer;
    51→  }
    52→
    53→  async validate(): Promise<ValidationResult> {
    54→    const errors: string[] = [];
    55→    const warnings: string[] = [];
    56→    const conflicts: Conflict[] = [];
    57→
    58→    try {
    59→      // Extract and validate ZIP structure
    60→      await this.extractZipContents();
    61→
    62→      // Validate metadata
    63→      const metadataValidation = await this.validateMetadata();
    64→      if (!metadataValidation.isValid) {
    65→        errors.push(...metadataValidation.errors);
    66→        return this.createValidationResult(false, conflicts, errors, warnings);
    67→      }
    68→
    69→      // Validate data files
    70→      const dataValidation = await this.validateDataFiles();
    71→      if (!dataValidation.isValid) {
    72→        errors.push(...dataValidation.errors);
    73→        warnings.push(...dataValidation.warnings);
    74→      }
    75→
    76→      // Detect conflicts with existing data
    77→      const conflictDetection = await this.detectConflicts();
    78→      conflicts.push(...conflictDetection.conflicts);
    79→      warnings.push(...conflictDetection.warnings);
    80→
    81→      const isValid = errors.length === 0;
    82→      return this.createValidationResult(isValid, conflicts, errors, warnings);
    83→    } catch (error) {
    84→      console.error("Backup validation error:", error);
    85→      errors.push(
    86→        `Validation failed: ${error instanceof Error ? error.message : "Unknown error"}`
    87→      );
    88→      return this.createValidationResult(false, conflicts, errors, warnings);
    89→    }
    90→  }
    91→
    92→  private async extractZipContents(): Promise<void> {
    93→    return new Promise((resolve, reject) => {
    94→      const directory = unzipper.Open.buffer(this.zipBuffer);
    95→
    96→      directory
    97→        .then(async (dir) => {
    98→          for (const file of dir.files) {
    99→            if (file.type === "File") {
   100→              const content = await file.buffer();
   101→
   102→              if (file.path.endsWith(".json")) {
   103→                try {
   104→                  const jsonData = JSON.parse(content.toString());
   105→                  this.extractedFiles.set(file.path, jsonData);
   106→                } catch (error) {
   107→                  throw new Error(
   108→                    `Invalid JSON in file ${file.path}: ${error}`
   109→                  );
   110→                }
   111→              } else {
   112→                // Store binary files (photos) as buffers
   113→                this.extractedFiles.set(file.path, content);
   114→              }
   115→            }
   116→          }
   117→          resolve();
   118→        })
   119→        .catch(reject);
   120→    });
   121→  }
   122→
   123→  private async validateMetadata(): Promise<{
   124→    isValid: boolean;
   125→    errors: string[];
   126→  }> {
   127→    const errors: string[] = [];
   128→
   129→    if (!this.extractedFiles.has("metadata.json")) {
   130→      errors.push("Missing metadata.json file");
   131→      return { isValid: false, errors };
   132→    }
   133→
   134→    try {
   135→      const metadataData = this.extractedFiles.get("metadata.json");
   136→      this.metadata = backupMetadataSchema.parse(metadataData);
   137→
   138→      // Check version compatibility
   139→      if (!SUPPORTED_VERSIONS.includes(this.metadata.version)) {
   140→        errors.push(
   141→          `Unsupported backup version: ${this.metadata.version}. Supported versions: ${SUPPORTED_VERSIONS.join(", ")}`
   142→        );
   143→      }
   144→
   145→      // Validate required data files exist
   146→      for (const dataFile of this.metadata.dataFiles) {
   147→        if (!this.extractedFiles.has(dataFile)) {
   148→          errors.push(`Missing required data file: ${dataFile}`);
   149→        }
   150→      }
   151→
   152→      return { isValid: errors.length === 0, errors };
   153→    } catch (error) {
   154→      errors.push(
   155→        `Invalid metadata format: ${error instanceof Error ? error.message : "Unknown error"}`
   156→      );
   157→      return { isValid: false, errors };
   158→    }
   159→  }
   160→
   161→  private async validateDataFiles(): Promise<{
   162→    isValid: boolean;
   163→    errors: string[];
   164→    warnings: string[];
   165→  }> {
   166→    const errors: string[] = [];
   167→    const warnings: string[] = [];
   168→
   169→    if (!this.metadata) {
   170→      errors.push("Metadata not loaded");
   171→      return { isValid: false, errors, warnings };
   172→    }
   173→
   174→    // Validate people data
   175→    if (this.extractedFiles.has("data/people.json")) {
   176→      const peopleData = this.extractedFiles.get("data/people.json");
   177→      if (!Array.isArray(peopleData)) {
   178→        errors.push("People data must be an array");
   179→      } else {
   180→        for (let i = 0; i < peopleData.length; i++) {
   181→          try {
   182→            personDataSchema.parse(peopleData[i]);
   183→          } catch (error) {
   184→            errors.push(
   185→              `Invalid person data at index ${i}: ${error instanceof Error ? error.message : "Unknown error"}`
   186→            );
   187→          }
   188→        }
   189→      }
   190→    }
   191→
   192→    // Validate users data
   193→    if (this.extractedFiles.has("data/users.json")) {
   194→      const usersData = this.extractedFiles.get("data/users.json");
   195→      if (!Array.isArray(usersData)) {
   196→        errors.push("Users data must be an array");
   197→      } else {
   198→        for (let i = 0; i < usersData.length; i++) {
   199→          try {
   200→            userDataSchema.parse(usersData[i]);
   201→          } catch (error) {
   202→            errors.push(
   203→              `Invalid user data at index ${i}: ${error instanceof Error ? error.message : "Unknown error"}`
   204→            );
   205→          }
   206→        }
   207→      }
   208→    }
   209→
   210→    // Validate relationships data
   211→    if (this.extractedFiles.has("data/relationships.json")) {
   212→      const relationshipsData = this.extractedFiles.get(
   213→        "data/relationships.json"
   214→      );
   215→      if (!Array.isArray(relationshipsData)) {
   216→        errors.push("Relationships data must be an array");
   217→      } else {
   218→        for (let i = 0; i < relationshipsData.length; i++) {
   219→          try {
   220→            relationshipDataSchema.parse(relationshipsData[i]);
   221→          } catch (error) {
   222→            errors.push(
   223→              `Invalid relationship data at index ${i}: ${error instanceof Error ? error.message : "Unknown error"}`
   224→            );
   225→          }
   226→        }
   227→      }
   228→    }
   229→
   230→    // Validate photo files
   231→    const photoCount = this.metadata.statistics.totalPhotos;
   232→    const actualPhotoFiles = Array.from(this.extractedFiles.keys()).filter(
   233→      (path) => path.startsWith("photos/") && !path.endsWith("/")
   234→    );
   235→
   236→    if (photoCount > 0 && actualPhotoFiles.length !== photoCount) {
   237→      warnings.push(
   238→        `Expected ${photoCount} photos but found ${actualPhotoFiles.length} photo files`
   239→      );
   240→    }
   241→
   242→    return { isValid: errors.length === 0, errors, warnings };
   243→  }
   244→
   245→  private async detectConflicts(): Promise<{
   246→    conflicts: Conflict[];
   247→    warnings: string[];
   248→  }> {
   249→    const conflicts: Conflict[] = [];
   250→    const warnings: string[] = [];
   251→
   252→    if (!this.metadata) {
   253→      warnings.push("Cannot detect conflicts without metadata");
   254→      return { conflicts, warnings };
   255→    }
   256→
   257→    // Check for existing people conflicts
   258→    if (this.extractedFiles.has("data/people.json")) {
   259→      const peopleData = this.extractedFiles.get("data/people.json");
   260→      const peopleConflicts = await this.detectPeopleConflicts(peopleData);
   261→      conflicts.push(...peopleConflicts);
   262→    }
   263→
   264→    // Check for existing users conflicts
   265→    if (this.extractedFiles.has("data/users.json")) {
   266→      const usersData = this.extractedFiles.get("data/users.json");
   267→      const userConflicts = await this.detectUserConflicts(usersData);
   268→      conflicts.push(...userConflicts);
   269→    }
   270→
   271→    // Check for relationship conflicts
   272→    if (this.extractedFiles.has("data/relationships.json")) {
   273→      const relationshipsData = this.extractedFiles.get(
   274→        "data/relationships.json"
   275→      );
   276→      const relationshipConflicts =
   277→        await this.detectRelationshipConflicts(relationshipsData);
   278→      conflicts.push(...relationshipConflicts);
   279→    }
   280→
   281→    return { conflicts, warnings };
   282→  }
   283→
   284→  private async detectPeopleConflicts(peopleData: any[]): Promise<Conflict[]> {
   285→    const conflicts: Conflict[] = [];
   286→
   287→    for (const person of peopleData) {
   288→      // Check for ID conflicts
   289→      const existingById = await db.person.findUnique({
   290→        where: { id: person.id },
   291→      });
   292→
   293→      if (existingById) {
   294→        conflicts.push({
   295→          type: "person",
   296→          action: "update",
   297→          existingId: existingById.id,
   298→          existingData: existingById,
   299→          newData: person,
   300→          conflictFields: this.getChangedFields(existingById, person),
   301→          severity: "medium",
   302→          description: `Person with ID ${person.id} already exists`,
   303→        });
   304→        continue;
   305→      }
   306→
   307→      // Check for email conflicts
   308→      if (person.email) {
   309→        const existingByEmail = await db.person.findFirst({
   310→          where: { email: person.email },
   311→        });
   312→
   313→        if (existingByEmail) {
   314→          conflicts.push({
   315→            type: "person",
   316→            action: "create",
   317→            existingId: existingByEmail.id,
   318→            existingData: existingByEmail,
   319→            newData: person,
   320→            conflictFields: ["email"],
   321→            severity: "high",
   322→            description: `Person with email ${person.email} already exists`,
   323→          });
   324→        }
   325→      }
   326→
   327→      // Check for name conflicts (potential duplicates)
   328→      const existingByName = await db.person.findFirst({
   329→        where: {
   330→          firstName: person.firstName,
   331→          lastName: person.lastName,
   332→          dateOfBirth: person.dateOfBirth
   333→            ? new Date(person.dateOfBirth)
   334→            : undefined,
   335→        },
   336→      });
   337→
   338→      if (existingByName) {
   339→        conflicts.push({
   340→          type: "person",
   341→          action: "create",
   342→          existingId: existingByName.id,
   343→          existingData: existingByName,
   344→          newData: person,
   345→          conflictFields: ["firstName", "lastName", "dateOfBirth"],
   346→          severity: "low",
   347→          description: `Potential duplicate person: ${person.firstName} ${person.lastName}`,
   348→        });
   349→      }
   350→    }
   351→
   352→    return conflicts;
   353→  }
   354→
   355→  private async detectUserConflicts(usersData: any[]): Promise<Conflict[]> {
   356→    const conflicts: Conflict[] = [];
   357→
   358→    for (const user of usersData) {
   359→      // Check for ID conflicts
   360→      const existingById = await db.user.findUnique({
   361→        where: { id: user.id },
   362→      });
   363→
   364→      if (existingById) {
   365→        conflicts.push({
   366→          type: "user",
   367→          action: "update",
   368→          existingId: existingById.id,
   369→          existingData: existingById,
   370→          newData: user,
   371→          conflictFields: this.getChangedFields(existingById, user),
   372→          severity: "high",
   373→          description: `User with ID ${user.id} already exists`,
   374→        });
   375→        continue;
   376→      }
   377→
   378→      // Check for email conflicts
   379→      const existingByEmail = await db.user.findUnique({
   380→        where: { email: user.email },
   381→      });
   382→
   383→      if (existingByEmail) {
   384→        conflicts.push({
   385→          type: "user",
   386→          action: "create",
   387→          existingId: existingByEmail.id,
   388→          existingData: existingByEmail,
   389→          newData: user,
   390→          conflictFields: ["email"],
   391→          severity: "high",
   392→          description: `User with email ${user.email} already exists`,
   393→        });
   394→      }
   395→    }
   396→
   397→    return conflicts;
   398→  }
   399→
   400→  private async detectRelationshipConflicts(
   401→    relationshipsData: any[]
   402→  ): Promise<Conflict[]> {
   403→    const conflicts: Conflict[] = [];
   404→
   405→    for (const relationship of relationshipsData) {
   406→      // Check for ID conflicts
   407→      const existingById = await db.relationship.findUnique({
   408→        where: { id: relationship.id },
   409→      });
   410→
   411→      if (existingById) {
   412→        conflicts.push({
   413→          type: "relationship",
   414→          action: "update",
   415→          existingId: existingById.id,
   416→          existingData: existingById,
   417→          newData: relationship,
   418→          conflictFields: this.getChangedFields(existingById, relationship),
   419→          severity: "medium",
   420→          description: `Relationship with ID ${relationship.id} already exists`,
   421→        });
   422→        continue;
   423→      }
   424→
   425→      // Check for duplicate relationships
   426→      const existingRelationship = await db.relationship.findFirst({
   427→        where: {
   428→          personId: relationship.personId,
   429→          relatedPersonId: relationship.relatedPersonId,
   430→          type: relationship.type,
   431→        },
   432→      });
   433→
   434→      if (existingRelationship) {
   435→        conflicts.push({
   436→          type: "relationship",
   437→          action: "create",
   438→          existingId: existingRelationship.id,
   439→          existingData: existingRelationship,
   440→          newData: relationship,
   441→          conflictFields: ["personId", "relatedPersonId", "type"],
   442→          severity: "medium",
   443→          description: `Duplicate relationship between ${relationship.personId} and ${relationship.relatedPersonId}`,
   444→        });
   445→      }
   446→    }
   447→
   448→    return conflicts;
   449→  }
   450→
   451→  private getChangedFields(existing: any, incoming: any): string[] {
   452→    const changedFields: string[] = [];
   453→
   454→    for (const key in incoming) {
   455→      if (existing[key] !== incoming[key]) {
   456→        changedFields.push(key);
   457→      }
   458→    }
   459→
   460→    return changedFields;
   461→  }
   462→
   463→  private createValidationResult(
   464→    isValid: boolean,
   465→    conflicts: Conflict[],
   466→    errors: string[],
   467→    warnings: string[]
   468→  ): ValidationResult {
   469→    const conflictsByType: Record<string, number> = {};
   470→    const conflictsBySeverity: Record<string, number> = {};
   471→
   472→    for (const conflict of conflicts) {
   473→      conflictsByType[conflict.type] =
   474→        (conflictsByType[conflict.type] || 0) + 1;
   475→      conflictsBySeverity[conflict.severity] =
   476→        (conflictsBySeverity[conflict.severity] || 0) + 1;
   477→    }
   478→
   479→    return {
   480→      isValid,
   481→      metadata: this.metadata!,
   482→      conflicts,
   483→      statistics: {
   484→        totalConflicts: conflicts.length,
   485→        conflictsByType,
   486→        conflictsBySeverity,
   487→      },
   488→      errors,
   489→      warnings,
   490→    };
   491→  }
   492→
   493→  getExtractedFiles(): Map<string, any> {
   494→    return this.extractedFiles;
   495→  }
   496→
   497→  getMetadata(): BackupMetadata | null {
   498→    return this.metadata;
   499→  }
