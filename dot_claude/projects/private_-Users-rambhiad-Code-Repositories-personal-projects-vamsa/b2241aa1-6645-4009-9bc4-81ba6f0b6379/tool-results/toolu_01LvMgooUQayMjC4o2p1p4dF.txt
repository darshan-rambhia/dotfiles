   352→async function importPhotosCore(
   353→  extractedFiles: Map<string, any>,
   354→  deps: RestoreDependencies
   355→): Promise<number> {
   356→  const storage = deps.getStorageAdapter();
   357→  let importedCount = 0;
   358→
   359→  for (const [filePath, fileBuffer] of extractedFiles.entries()) {
   360→    if (!filePath.startsWith("photos/") || filePath.endsWith("/")) {
   361→      continue;
   362→    }
   363→
   364→    try {
   365→      // Extract person ID and filename from path: photos/{personId}/{filename}
   366→      const pathParts = filePath.split("/");
   367→      if (pathParts.length !== 3) continue;
   368→
   369→      const personId = pathParts[1];
   370→      const filename = pathParts[2];
   371→
   372→      // Check if person exists
   373→      const person = await deps.db.person.findUnique({
   374→        where: { id: personId },
   375→      });
   376→      if (!person) {
   377→        console.warn(`Skipping photo for non-existent person: ${personId}`);
   378→        continue;
   379→      }
   380→
   381→      // Upload photo to storage
   382→      const storedPath = await storage.upload(fileBuffer, filename, "image/jpeg");
   383→      const photoUrl = storage.getUrl(storedPath);
   384→
   385→      // Update person with photo URL
   386→      await deps.db.person.update({
   387→        where: { id: personId },
   388→        data: { photoUrl },
   389→      });
   390→
   391→      importedCount++;
   392→    } catch (error) {
   393→      console.warn(`Failed to import photo ${filePath}:`, error);
   394→    }
   395→  }
   396→
   397→  return importedCount;
   398→}
